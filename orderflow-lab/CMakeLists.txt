# apps
function(ofl_add_app name)
  add_executable(${name} apps/${name}.cpp)
  qc_target_defaults(${name})
  target_link_libraries(${name} PRIVATE ofl qc)
  target_compile_definitions(${name} PRIVATE OFL_RESULTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/results")
endfunction()
